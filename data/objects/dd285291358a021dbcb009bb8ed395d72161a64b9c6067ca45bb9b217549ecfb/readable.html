(function() {
    'use strict';

    function initHeader13Categories() {
        var swiperContainer = document.querySelector('.header-13 .header-13-categories-swiper');
        if (!swiperContainer) return;

        var categoriesSwiper = new Swiper('.header-13 .header-13-categories-swiper', {
            slidesPerView: 'auto',
            spaceBetween: 0,
            loop: false,
            freeMode: {
                enabled: true,
                sticky: false
            },
            mousewheel: {
                forceToAxis: true
            },
            grabCursor: true,
            navigation: {
                nextEl: '.header-13 .header-13-swiper-button-next',
                prevEl: '.header-13 .header-13-swiper-button-prev'
            },
            on: {
                init: function() {
                    updateNavigationButtons(this);
                },
                resize: function() {
                    this.update();
                    updateNavigationButtons(this);
                },
                setTranslate: function() {
                    updateNavigationButtons(this);
                }
            }
        });

        function updateNavigationButtons(swiper) {
            var prevButton = document.querySelector('.header-13 .header-13-swiper-button-prev');
            var nextButton = document.querySelector('.header-13 .header-13-swiper-button-next');
            if (!prevButton || !nextButton) return;

            var totalWidth = 0;
            swiper.slides.forEach(function(slide) {
                if (!slide.classList.contains('swiper-slide-duplicate')) {
                    totalWidth += slide.offsetWidth;
                }
            });

            var containerWidth = swiper.el.offsetWidth;

            if (totalWidth <= containerWidth) {
                prevButton.classList.remove('swiper-button-active');
                nextButton.classList.remove('swiper-button-active');
            } else {
                if (swiper.isBeginning) {
                    prevButton.classList.remove('swiper-button-active');
                } else {
                    prevButton.classList.add('swiper-button-active');
                }
                if (swiper.isEnd) {
                    nextButton.classList.remove('swiper-button-active');
                } else {
                    nextButton.classList.add('swiper-button-active');
                }
            }
        }

        var resizeTimer;
        window.addEventListener('resize', function() {
            clearTimeout(resizeTimer);
            resizeTimer = setTimeout(function() {
                categoriesSwiper.update();
                updateNavigationButtons(categoriesSwiper);
            }, 250);
        });
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initHeader13Categories);
    } else {
        initHeader13Categories();
    }
})();
</script>

    var shareableModelId = 26799194;

    function shareCount(id, model, platform, url) {
        fetch("https://www.milligazete.com.tr/sharecount", {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]')?.getAttribute('content')
            },
            body: JSON.stringify({ id, model, platform, url })
        }).catch(err => console.error('Share count fetch error:', err));

        bumpShareCounter();
    }

    // Paylaşım sayacını yerelde +1 göster. Kalıcı değer edge /v1/hit yanıtından
    // gelir ama o taban 15dk cache'li — kullanıcı kendi paylaşımının sayıya
    // yansıdığını hemen görsün.
    //
    // Sayaç tire (—) ise, yani değer 0 ya da henüz gelmemişse, ilk paylaşım
    // onu "1" yapar. statsRaw dataset'ten okunur (tire metninden sayı
    // üretilemez); yoksa 0 varsayılır.
    function bumpShareCounter() {
        const box = document.querySelector('[data-stats="share"]');
        if (!box) {
            return;
        }
        const target = box.querySelector('[data-stats-value="share"]');
        if (!target) {
            return;
        }
        const current = parseInt(box.dataset.statsRaw || '0', 10) || 0;
        const next = current + 1;
        box.dataset.statsRaw = String(next);
        target.textContent = window.HitManagement
            ? new window.HitManagement().formatShort(next)
            : String(next);
    }

    function goSharePopup(url, title, width = 600, height = 400) {
        const left = (screen.width - width) / 2;
        const top = (screen.height - height) / 2;
        window.open(
            url,
            title,
            `width=${width},height=${height},left=${left},top=${top},resizable=yes,scrollbars=yes`
        );
    }

    async function handleMobileShare(event, title, url) {
        event.preventDefault();

        if (shareableModelId && shareableModelClass) {
            shareCount(shareableModelId, shareableModelClass, 'native_mobile_share', url);
        }

        const isAndroidWebView = navigator.userAgent.includes('Android') && !navigator.share;

        if (isAndroidWebView) {
            window.location.href = 'androidshare://paylas?title=' + encodeURIComponent(title) + '&url=' + encodeURIComponent(url);
            return;
        }

        if (navigator.share) {
            try {
                await navigator.share({ title: title, url: url });
            } catch (error) {
                if (error.name !== 'AbortError') {
                    console.error('Web Share API failed:', error);
                }
            }
        } else {
            alert("Bu cihaz paylaşımı desteklemiyor.");
        }
    }

    function initiateDesktopShare(event, platformOverride = null) {
        event.preventDefault();
        const anchor = event.currentTarget;
        const platform = platformOverride || anchor.dataset.platform;
        const webShareUrl = anchor.href;
        const contentUrl = anchor.dataset.shareUrl || webShareUrl;

        if (shareableModelId && shareableModelClass && platform) {
            shareCount(shareableModelId, shareableModelClass, platform, contentUrl);
        }

        goSharePopup(webShareUrl, platform ? platform.charAt(0).toUpperCase() + platform.slice(1) : "Share");
    }

    function copyURL(event, urlToCopy) {
        event.preventDefault();
        navigator.clipboard.writeText(urlToCopy).then(() => {
            alert('Bağlantı panoya kopyalandı!');
        }).catch(err => {
            console.error('Could not copy text: ', err);
            try {
                const textArea = document.createElement("textarea");
                textArea.value = urlToCopy;
                textArea.style.position = "fixed";
                document.body.appendChild(textArea);
                textArea.focus();
                textArea.select();
                document.execCommand('copy');
                document.body.removeChild(textArea);
                alert('Bağlantı panoya kopyalandı!');
            } catch (fallbackErr) {
                console.error('Fallback copy failed:', fallbackErr);
            }
        });
    }

    function printContent(event) {
        event.preventDefault();

        const triggerElement = event.currentTarget;
        const contextContainer = triggerElement.closest('.infinite-item') || document;

        const articleBody = contextContainer.querySelector('.article-text');

            return;

        let printHtml = '';
        
        if (header) {
            const titleEl = header.querySelector('h1');
            const descEl  = header.querySelector('.description, h2.lead');

            let cleanHeaderHtml = '<div class="printed-header">';
            if (titleEl) cleanHeaderHtml += titleEl.outerHTML;
            if (descEl)  cleanHeaderHtml += descEl.outerHTML;
            cleanHeaderHtml += '</div>';

            printHtml += cleanHeaderHtml;
        }

        if (media) {
            printHtml += media.outerHTML;
        }

        if (articleBody) {
            const articleClone = articleBody.cloneNode(true);
            articleClone.querySelectorAll('.post-flash').forEach(function (el) {
                el.parentNode.removeChild(el);

            printHtml += articleClone.outerHTML;
        }
        const iframe = document.createElement('iframe');
        iframe.style.position = 'fixed';
        iframe.style.right = '0';
        iframe.style.bottom = '0';
        iframe.style.width = '0';
        iframe.style.height = '0';
        iframe.style.border = '0';
        document.body.appendChild(iframe);

        const title = document.title || 'Yazdır';

        // Yazdırma belgesi DOM API ile kurulur. Eskiden doc.write ile tam bir HTML
        // belgesi metni yazılıyordu; o metin sayfa kaynağında fazladan head/body

        doc.close();

        doc.documentElement.lang = 'tr';

        const charset = doc.createElement('meta');
        charset.setAttribute('charset', 'utf-8');
        doc.head.appendChild(charset);

        const styleLoads = [];
        document.querySelectorAll('link[rel="stylesheet"], style').forEach(function (el) {
            const clone = doc.importNode(el, true);
            if (clone.tagName === 'LINK') {
                styleLoads.push(new Promise(function (resolve) {
                    clone.onload = clone.onerror = resolve;
                }));
            }
            doc.head.appendChild(clone);
        });

        const printStyle = doc.createElement('style');
        printStyle.textContent =
            'html, body { margin: 0; padding: 0; background: #ffffff; }' +
            '.printed-article { margin: 0; padding: 20px; box-shadow: none; background: #ffffff; }';
        doc.head.appendChild(printStyle);

        const wrapper = doc.createElement('div');
        wrapper.className = 'printed-article';
        wrapper.innerHTML = printHtml;
        doc.body.appendChild(wrapper);

        // Stil dosyası takılırsa yazdırma sonsuza dek beklemesin
        const stylesReady = Promise.race([
            Promise.all(styleLoads),
            new Promise(function (resolve) { setTimeout(resolve, 3000); })

                frameWindow.print();
            } finally {

                    document.body.removeChild(iframe);

            }
        });

<script>
(function() {
    document.querySelectorAll('[data-widget-unique-key="namaz_vakitleri_18790"]:not([data-prayer-init])').forEach(function(widget) {
        widget.setAttribute('data-prayer-init', '1');
        var el = widget.querySelector('.remaining');
        if (!el) return;

        var items = widget.querySelectorAll('.time-item');
        var times = [];
        items.forEach(function(item) {
            var val = item.querySelector('.time-value').textContent.trim();
            var p = val.split(':');
            times.push(parseInt(p[0]) * 60 + parseInt(p[1]));
        });

        function getNow() {
            var d = new Date();
            return { minutes: d.getHours() * 60 + d.getMinutes(), seconds: d.getSeconds() };
        }

        function findCurrentAndNext() {
            var now = getNow();
            var currentIdx = -1, nextIdx = -1;
            for (var i = 0; i < times.length; i++) {
                if (now.minutes >= times[i]) currentIdx = i;
                else if (nextIdx === -1) nextIdx = i;
            }
            if (nextIdx === -1) nextIdx = 0;
            return { current: currentIdx, next: nextIdx };
        }

        function updateStates() {
            var state = findCurrentAndNext();
            items.forEach(function(item, i) {
                item.classList.remove('passed', 'active', 'upcoming');
                var icon = item.querySelector('.time-icon');
                if (i === state.current) {
                    item.classList.add('active');
                    icon.innerHTML = '<i class="fa fa-circle"></i>';
                } else if (i < state.current || (state.current === -1 && i < times.length)) {
                    item.classList.add('passed');
                    icon.innerHTML = '<i class="fa fa-check"></i>';
                } else {
                    item.classList.add('upcoming');
                    icon.innerHTML = '<i class="fa fa-hourglass-half"></i>';
                }
            });

            var nameEl = widget.querySelector('.next-prayer-name');
            if (nameEl) {
                var nextName = items[state.next].querySelector('.time-name').textContent.trim();
                nameEl.innerHTML = '<i class="fa fa-clock me-1"></i>' + nextName + ' Namazı';
            }
        }

        function startCountdown() {
            var state = findCurrentAndNext();
            var now = getNow();
            var targetMin = times[state.next];
            var diffSec = (targetMin - now.minutes) * 60 - now.seconds;
            if (diffSec <= 0) diffSec += 86400;

            var h = Math.floor(diffSec / 3600);
            var m = Math.floor((diffSec % 3600) / 60);
            var s = diffSec % 60;


            if (widget._prayerInterval) clearInterval(widget._prayerInterval);

            el.textContent = h > 0
                ? String(h).padStart(2,'0') + ':' + String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0')
                : String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0');

            widget._prayerInterval = setInterval(function() {
                if (s > 0) { s--; }
                else if (m > 0) { m--; s = 59; }
                else if (h > 0) { h--; m = 59; s = 59; }
                else {
                    clearInterval(widget._prayerInterval);
                    updateStates();
                    startCountdown();
                    return;
                }
                el.textContent = h > 0
                    ? String(h).padStart(2,'0') + ':' + String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0')
                    : String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0');
            }, 1000);
        }

        updateStates();
        startCountdown();
    });
})();
