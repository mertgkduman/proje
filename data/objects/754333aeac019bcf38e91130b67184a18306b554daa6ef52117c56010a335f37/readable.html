    </div>
    <div class="actions">
      <a class="btn btn-back" id="backBtn" href="https://saadet.org.tr/dosya-arsivi">← Geri Dön</a>
      <a class="btn btn-open" id="openBtn" href="#" target="_blank" rel="noopener">PDF'yi Yeni Sekmede Aç</a>

  <div class="stage">
    <div class="status" id="status">Sayfalar hazırlanıyor...</div>
    <div id="pages"></div>
    <div class="sentinel" id="sentinel">Kaydırarak devam edin</div>
  </div>
  <script>
    (function(){
      var params=new URLSearchParams(location.search);
      var file=params.get('f')||params.get('file')||'';
      var title=params.get('t')||params.get('title')||'Belge Önizleme';
      document.getElementById('title').textContent=title;
      document.getElementById('backBtn').onclick=function(e){ if(history.length>1){ e.preventDefault(); history.back(); } };
      if(!file){ document.getElementById('status').textContent='Dosya bulunamadı'; return; }
      var name=String(file).replace(/\\/g,'/').split('/').pop();
      if(!name || name.indexOf('..')!==-1 || !/\.pdf$/i.test(name)){ document.getElementById('status').textContent='Geçersiz dosya'; return; }

      var pdfUrl='/dosyalar/'+encodeURIComponent(name);
      document.getElementById('openBtn').href=pdfUrl;

      var total=0, next=1, loading=false, done=false, failStreak=0;
      var pagesEl=document.getElementById('pages');
      var statusEl=document.getElementById('status');
      var sentinel=document.getElementById('sentinel');
      var FIRST=8, BATCH=4;

      function pageUrl(p){ return 'page.php?f='+encodeURIComponent(name)+'&p='+p+'&v=4'; }

      function addPage(p, src){
        if(document.getElementById('page-'+p)) return;
        var wrap=document.createElement('div');
        wrap.className='page';
        wrap.id='page-'+p;
        wrap.innerHTML='<div class="num">Sayfa '+p+(total?(' / '+total):'')+'</div>';
        var img=document.createElement('img');
        img.alt=title+' - sayfa '+p;
        img.decoding='async';
        img.src=src;
        wrap.appendChild(img);
        var kids=pagesEl.children;
        var placed=false;
        for(var i=0;i<kids.length;i++){
          var n=parseInt(String(kids[i].id||'').replace('page-',''),10);
          if(n>p){ pagesEl.insertBefore(wrap, kids[i]); placed=true; break; }
        }
        if(!placed) pagesEl.appendChild(wrap);
      }

      function loadOne(p){
        return new Promise(function(resolve){
          var img=new Image();
          var settled=false;
          var timer=setTimeout(function(){ if(!settled){ settled=true; resolve(false); } }, 45000);
          img.onload=function(){ if(settled) return; settled=true; clearTimeout(timer); addPage(p, img.src); resolve(true); };
          img.onerror=function(){ if(settled) return; settled=true; clearTimeout(timer); resolve(false); };
          img.src=pageUrl(p);
        });
      }

      async function loadMore(){
        if(loading || done) return;
        loading=true;
        sentinel.textContent='Sayfalar yükleniyor...';
        var batch=(next===1?FIRST:BATCH);
        var start=next;
        var jobs=[];
        for(var i=0;i<batch;i++){
          var p=start+i;
          if(total && p>total) break;
          jobs.push(loadOne(p));
        }
        if(!jobs.length){ done=true; loading=false; sentinel.textContent=total?('Toplam '+total+' sayfa yüklendi.'):'Belge sonu.'; return; }
        var results=await Promise.all(jobs);
        var okCount=0;
        for(var j=0;j<results.length;j++){
          if(results[j]){ okCount++; failStreak=0; next=start+j+1; }
          else {
            failStreak++;
            if(total && (start+j)>total){ done=true; break; }
            if(!total && failStreak>=2){ done=true; break; }
            if(total && (start+j)<=total){
              // known page failed: stop advancing past hole but allow retry on next scroll
              next=start+j;
              break;
            }
          }
        }
        if(okCount){
          statusEl.textContent= total
            ? ('Sayfa '+(next-1)+' / '+total+' gösteriliyor. Aşağı kaydırarak devam edin.')
            : ('Sayfa '+(next-1)+' gösteriliyor. Aşağı kaydırarak devam edin.');
        }
        loading=false;
        if(done){
          sentinel.textContent= total ? ('Toplam '+total+' sayfa. Kaydırarak okuyabilirsiniz.') : 'Belge sonu.';
        } else {
          sentinel.textContent='Kaydırarak devam edin';
          maybeLoad();
        }
      }

      function maybeLoad(){
        if(done || loading) return;
        var rect=sentinel.getBoundingClientRect();
        if(rect.top < window.innerHeight + 1200) loadMore();
      }

      fetch('meta.php?f='+encodeURIComponent(name)).then(function(r){return r.json()}).then(function(j){
        if(j && j.ok && j.pages){ total=j.pages; statusEl.textContent='Toplam '+total+' sayfa. Sayfalar açılıyor...'; }
      }).catch(function(){}).finally(function(){ loadMore(); });

      window.addEventListener('scroll', maybeLoad, {passive:true});
      window.addEventListener('resize', maybeLoad);
    })();
  </script>
