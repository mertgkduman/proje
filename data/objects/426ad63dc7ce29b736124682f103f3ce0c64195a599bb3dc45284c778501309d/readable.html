<script>
    function toggleMobileAccordion(element) {
        const accordionItem = element.closest('.mobile-accordion-item');
        const accordionContent = accordionItem.querySelector('.mobile-accordion-content');
        const isActive = element.classList.contains('active');

        // Tüm accordion'ları kapat
        document.querySelectorAll('.mobile-accordion-toggle').forEach(function(toggle) {
            toggle.classList.remove('active');
        });
        document.querySelectorAll('.mobile-accordion-content').forEach(function(content) {
            content.classList.remove('active');
        });

        // Eğer tıklanan accordion zaten açık değilse, onu aç
        if (!isActive) {
            element.classList.add('active');
            accordionContent.classList.add('active');
        }
    }

    function toggleMobileSubAccordion(element) {
        const subItem = element.closest('.mobile-accordion-sub-item');
        const subContent = subItem.querySelector('.mobile-accordion-sub-content');
        const isActive = element.classList.contains('active');

        // Aynı parent içindeki diğer sub-accordion'ları kapat
        const parentAccordion = element.closest('.mobile-accordion-content');
        if (parentAccordion) {
            parentAccordion.querySelectorAll('.mobile-accordion-sub-toggle').forEach(function(toggle) {
                if (toggle !== element) {
                    toggle.classList.remove('active');
                }
            });
            parentAccordion.querySelectorAll('.mobile-accordion-sub-content').forEach(function(content) {
                if (content !== subContent) {
                    content.classList.remove('active');
                }
            });
        }

        // Eğer tıklanan sub-accordion zaten açık değilse, onu aç
        if (!isActive) {
            element.classList.add('active');
            subContent.classList.add('active');
        } else {
            element.classList.remove('active');
            subContent.classList.remove('active');
        }
    }
</script>

<script>
    // Touch event listener'larını otomatik olarak passive yap (performans iyileştirmesi)
    (function() {
        if (typeof window.addEventListener === 'undefined') return;
        
        var supportsPassive = false;
        try {
            var opts = Object.defineProperty({}, 'passive', {
                get: function() {
                    supportsPassive = true;
                }
            });
            window.addEventListener('testPassive', null, opts);
            window.removeEventListener('testPassive', null, opts);
        } catch (e) {}
        
        if (!supportsPassive) return; // Eski tarayıcılar için çık
        
        // EventTarget.prototype.addEventListener'i override et
        var originalAddEventListener = EventTarget.prototype.addEventListener;
        EventTarget.prototype.addEventListener = function(type, listener, options) {
            // Touch event'leri için passive ekle (eğer zaten belirtilmemişse)
            if (type === 'touchstart' || type === 'touchmove') {
                if (typeof options === 'boolean') {
                    options = { capture: options, passive: true };
                } else if (typeof options === 'object' && options !== null) {
                    if (options.passive === undefined) {
                        options = Object.assign({}, options, { passive: true });
                    }
                } else {
                    options = { passive: true };
                }
            }
            return originalAddEventListener.call(this, type, listener, options);
        };
    })();
</script>

<script>
(function() {
    'use strict';
    
    // Placeholder path - config'den al veya sabit kullan
    const PLACEHOLDER_PATH = 'https://t24.com.tr/placeholder.png';
    
    // İşlenmiş görselleri takip et (sonsuz döngüyü önlemek için)
    const processedImages = new WeakSet();

    /**
     * URL'deki path segmentlerini güvenli şekilde encode et.
     * Özellikle "( )" ve boşluk içeren legacy medya URL'leri için.
     */
    function normalizeImageUrl(src) {
        if (!src || typeof src !== 'string') {
            return src;
        }

        // Sadece media-cdn görselleri için uygula.
        if (!src.includes("media-cdn.t24.com.tr")) {
            return src;
        }

        try {
            const urlObj = new URL(src, window.location.origin);
            const segments = urlObj.pathname.split('/').map(function(segment) {
                if (!segment) return segment;
                try {
                    return encodeURIComponent(decodeURIComponent(segment));
                } catch (e) {
                    return encodeURIComponent(segment);
                }
            });
            urlObj.pathname = segments.join('/');
            return urlObj.toString();
        } catch (e) {
            // URL parse edilemezse en azından boşlukları encode et.
            return src.replace(/ /g, '%20');
        }
    }
    
    /**
     * Görsel hatası için handler
     */
    function handleImageError(img) {
        // Eğer zaten işlendiyse veya placeholder ise, tekrar işleme
        if (processedImages.has(img) || img.src === PLACEHOLDER_PATH || img.src.includes('placeholder.png')) {
            return;
        }

        // Placeholder'a düşmeden önce normalize URL ile 1 kez yeniden dene.
        if (!img.dataset.urlNormalizedRetry) {
            const currentSrc = img.getAttribute('src') || img.src || '';
            const normalizedSrc = normalizeImageUrl(currentSrc);
            if (normalizedSrc && normalizedSrc !== currentSrc) {
                img.dataset.urlNormalizedRetry = '1';
                img.src = normalizedSrc;
                return;
            }
            img.dataset.urlNormalizedRetry = '1';
        }
        
        // İşlendi olarak işaretle
        processedImages.add(img);
        
        // Placeholder'a değiştir
        img.src = PLACEHOLDER_PATH;
        
        // Placeholder da yüklenemezse, error event'ini kaldır (sonsuz döngüyü önle)
        img.onerror = function() {
            this.onerror = null; // Event handler'ı kaldır
            this.style.display = 'none'; // Görseli gizle
        };
    }
    
    /**
     * Mevcut tüm görselleri işle
     */
    function processExistingImages() {
        const images = document.querySelectorAll('img:not([data-error-handled])');
        images.forEach(function(img) {
            // Zaten error handler varsa atla
            if (img.hasAttribute('data-error-handled')) {
                return;
            }
            
            // Attribute ekle (tekrar işlenmesini önle)
            img.setAttribute('data-error-handled', 'true');

            // İlk yüklemeden önce URL normalize et
            const originalSrc = img.getAttribute('src') || '';
            const normalizedSrc = normalizeImageUrl(originalSrc);
            if (normalizedSrc && normalizedSrc !== originalSrc) {
                img.setAttribute('src', normalizedSrc);
            }
            
            // Eğer görsel zaten yüklenmişse ve hata varsa kontrol et
            if (img.complete && img.naturalHeight === 0) {
                handleImageError(img);
            } else {
                // Error event handler ekle
                img.addEventListener('error', function() {
                    handleImageError(this);
                }, { once: true }); // once: true ile sadece bir kez çalışsın
            }
        });
    }
    
    /**
     * Dinamik eklenen görselleri yakala (MutationObserver)
     */
    function observeNewImages() {
        // MutationObserver sadece modern tarayıcılarda çalışır
        if (typeof MutationObserver === 'undefined') {
            return;
        }
        
        const observer = new MutationObserver(function(mutations) {
            mutations.forEach(function(mutation) {
                mutation.addedNodes.forEach(function(node) {
                    // Eğer node bir img ise
                    if (node.nodeType === 1 && node.tagName === 'IMG') {
                        if (!node.hasAttribute('data-error-handled')) {
                            node.setAttribute('data-error-handled', 'true');
                            node.addEventListener('error', function() {
                                handleImageError(this);
                            }, { once: true });
                        }
                    }
                    
                    // Eğer node içinde img'ler varsa
                    if (node.nodeType === 1 && node.querySelectorAll) {
                        const images = node.querySelectorAll('img:not([data-error-handled])');
                        images.forEach(function(img) {
                            img.setAttribute('data-error-handled', 'true');
                            img.addEventListener('error', function() {
                                handleImageError(this);
                            }, { once: true });
                        });
                    }
                });
            });
        });
        
        // Document body'yi gözlemle
        observer.observe(document.body, {
            childList: true,
            subtree: true
        });
    }
    
    // DOM yüklendiğinde çalıştır
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', function() {
            processExistingImages();
            observeNewImages();
        });
    } else {
        // DOM zaten yüklüyse hemen çalıştır
        processExistingImages();
        observeNewImages();
    }
})();
