<script>
    function emojiComponent(postId, initialEmojis) {
        return {
            emojis: initialEmojis.map(e => ({
                id: e.id,
                name: e.name,
                icon: e.icon,
                count: e.count,
                selected: e.selected,
                percentage: 0,
                loading: false,
            })),
            hovered: null,

            async getFreshCsrfToken() {
                try {
                    const res = await fetch('/csrf-token.js', { cache: 'no-store' });
                    const text = await res.text();
                    const match = text.match(/'(.+)'/);
                    return match ? match[1] : null;
                } catch (err) {
                    console.error('CSRF token alınamadı:', err);
                    return null;
                }
            },

            init() {
                this.updatePercentages();
            },

            async selectEmoji(emoji) {
                if (!emoji.selected && this.emojis.filter(e => e.selected).length >= 2) {
                    Swal.fire({
                        title: "Uyar\u0131",
                        text: "En fazla iki farkl\u0131 emojiye oy verebilirsiniz.",
                        icon: 'warning',
                        confirmButtonColor: '#d33',
                        timer: 2000,
                        showConfirmButton: false
                    });
                    return;
                }

                const previousState = emoji.selected;
                const previousCount = emoji.count;
                emoji.selected = !emoji.selected;
                this.hovered = null;
                emoji.loading = true;

                try {
                    await this.sendEmojiData(emoji, previousState, previousCount);
                    this.updatePercentages();
                } catch (error) {
                    emoji.selected = previousState;
                    emoji.count = previousCount;
                    this.updatePercentages();
                    Swal.fire({
                        title: "Hata",
                        text: "Bir hata olu\u015ftu, l\u00fctfen tekrar deneyin.",
                        icon: 'error',
                        confirmButtonColor: '#d33',
                        timer: 2000,
                        showConfirmButton: false
                    });
                } finally {
                    emoji.loading = false;
                }
            },

            async sendEmojiData(emoji, previousState, previousCount) {
                const csrfToken = await this.getFreshCsrfToken();
                if (!csrfToken) {
                    throw new Error("CSRF token al\u0131namad\u0131");
                }

                const response = await fetch(`/posts/${postId}/emojis`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': csrfToken,
                        'X-Requested-With': 'XMLHttpRequest'
                    },
                    body: JSON.stringify({
                        emoji_id: emoji.id,
                        post_id: postId,
                        selected: emoji.selected
                    })
                });

                const data = await response.json();

                if (!response.ok || data.error) {
                    Swal.fire({
                        title: "Uyar\u0131",
                        text: data.error || "Bir hata olu\u015ftu.",
                        icon: 'warning',
                        timer: 2000,
                        showConfirmButton: false
                    });
                    emoji.selected = previousState;
                    emoji.count = previousCount;
                    this.updatePercentages();
                    return;
                }

                if (data.success) {
                    emoji.count = data.total;
                    this.updatePercentages();
                }
            },

            updatePercentages() {
                const total = this.emojis.reduce((acc, cur) => acc + cur.count, 0);
                this.emojis.forEach(emoji => {
                    emoji.percentage = total > 0 ? (emoji.count / total) * 100 : 0;
                });
            }
        };
    }
