            if (isLiked) {
                localStorage.setItem(stateKey, '0');
                n = Math.max(0, n - 1);
                likeBtn.classList.remove('text-[#e30613]');
            } else {
                localStorage.setItem(stateKey, '1');
                n += 1;
                likeBtn.classList.add('text-[#e30613]');
            }
            localStorage.setItem(countKey, n);
            likeBtn.dataset.likeCount = n;
            countNode.textContent = n;
            return;
        }


        // Audio play/pause
        const audioBtn = event.target.closest('.article-audio-toggle');
        if (audioBtn) {
            if (activeButton === audioBtn) { stopAudio(); return; }
            stopAudio();

            const articleEl = audioBtn.closest('article') ?? document;
            const body    = articleEl.querySelector('.article-content-body')?.textContent?.trim() ?? '';
            const title   = articleEl.querySelector('h1, h2')?.textContent?.trim() ?? '';
            const summary = articleEl.querySelector('.article-summary')?.textContent?.trim() ?? '';
            if (!body) return;

            activePlainText  = [title, summary, body].filter(Boolean).join('. ');
            activeTotalSec   = Math.ceil(activePlainText.length / (CHARS_PER_SEC * activeRate));
            activeElapsedSec = 0;

            updateWidgetUI(audioBtn, 0, activeTotalSec);
            doSpeak(audioBtn, 0);
            return;
        }

        // -10s / +10s
        const skipTarget = event.target.closest('[data-audio-action]');
        if (skipTarget && activeButton) {
            const action = skipTarget.dataset.audioAction;
            if (action === 'back') { seekAndSpeak(activeButton, activeElapsedSec - 10); return; }
            if (action === 'fwd')  { seekAndSpeak(activeButton, activeElapsedSec + 10); return; }
        }

        // Speed
        const speedBtn = event.target.closest('.article-audio-speed');
        if (speedBtn) {
            const rates = [1, 1.25, 1.5, 2];
            const cur  = parseFloat(speedBtn.dataset.rate ?? '1');
            const next = rates[(rates.indexOf(cur) + 1) % rates.length];
            speedBtn.dataset.rate = next;
            speedBtn.textContent  = next + 'x';
            activeRate = next;
            if (activeButton) {
                const savedBtn  = activeButton;
                const savedText = activePlainText;
                clearTicker();
                activeButton = null;
                window.speechSynthesis.cancel();
                activePlainText  = savedText;
                activeTotalSec   = Math.ceil(savedText.length / (CHARS_PER_SEC * next));
                activeElapsedSec = 0;
                setTimeout(() => doSpeak(savedBtn, 0), 150);
            }
            return;
        }
    });

    if (streamContainer) observeSentinels();

    // Sayfadan ayrılınca sesi durdur
    const stopOnLeave = () => { if (activeButton) stopAudio(); };
    window.addEventListener('beforeunload', stopOnLeave);
    document.addEventListener('visibilitychange', () => { if (document.hidden && activeButton) stopAudio(); });
})();
