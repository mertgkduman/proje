<script>
/*
====================================================
KÖŞE YAZARLARI ALANI ÖZEL DÜZENLEME
----------------------------------------------------
1) Mobilde Owl Carousel devre dışı bırakılır
   ve köşe yazıları alt alta listelenir.
2) "Önder Yıldırımcan" mobil listede
   SADECE 1 KEZ ve EN BAŞTA gösterilir.
3) Masaüstü görünümüne ve slider yapısına
   KESİNLİKLE dokunulmaz.
----------------------------------------------------
Not: Bu kod DOM yüklendikten sonra çalışır.
====================================================
*/

document.addEventListener("DOMContentLoaded", function () {

    /* === SADECE MOBİL === */
    if (window.innerWidth > 768) return;

    var $carousel = $('.ne-carousel');

    /* Owl Carousel aktifse mobilde kapat */
    if ($carousel.hasClass('owl-loaded')) {
        $carousel.trigger('destroy.owl.carousel');
    }

    /* Clone edilmiş item'ları TAMAMEN kaldır */
    $carousel.find('.owl-item.cloned').remove();

    /* Owl sarmalayıcılarını kaldır */
    $carousel
        .removeClass('owl-carousel owl-loaded owl-theme')
        .find('.owl-stage-outer, .owl-stage')
        .children()
        .unwrap()
        .unwrap();

    /* owl-item class'larını temizle */
    $carousel.find('.owl-item').removeClass('owl-item');

    /* Oklar ve noktaları kaldır */
    $carousel.find('.owl-nav, .owl-dots').remove();

    /* Önder Yıldırımcan'ı TEK KOPYA olarak başa al */
    var $onder = $carousel.find('.hover-show-play-btn').filter(function () {
        return $(this).find('.yazarlar a').text().trim() === 'Önder Yıldırımcan';
    }).first();

    if ($onder.length) {
        $carousel.prepend($onder);
    }

});
</script>
