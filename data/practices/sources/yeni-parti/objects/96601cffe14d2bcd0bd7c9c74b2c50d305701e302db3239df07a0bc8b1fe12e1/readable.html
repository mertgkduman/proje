      var desc   = DESC[code] || 'Hava Durumu';
      var temp   = Math.round(d.current.temperature_2m);
      var hum    = Math.round(d.current.relative_humidity_2m);
      var wind   = Math.round(d.current.wind_speed_10m);
      var tmax   = Math.round(d.daily.temperature_2m_max[0]);
      var tmin   = Math.round(d.daily.temperature_2m_min[0]);
      var dayName = DAYS[new Date().getDay()];

      box.innerHTML =
        '<div class="weather-top">' +
          '<div class="weather-icon"><i class="fas ' + icon + '"></i></div>' +
          '<div class="weather-main">' +
            '<span class="weather-temp">' + temp + '°<small>C</small></span>' +
            '<span class="weather-city"><i class="fas fa-location-dot"></i> ' + CITY + '</span>' +
          '</div>' +
        '</div>' +
        '<div class="weather-desc">' + desc + ' &middot; ' + dayName + '</div>' +
        '<div class="weather-stats">' +
          '<div class="weather-stat"><i class="fas fa-arrow-up"></i> ' + tmax + '°</div>' +
          '<div class="weather-stat"><i class="fas fa-arrow-down"></i> ' + tmin + '°</div>' +
          '<div class="weather-stat"><i class="fas fa-droplet"></i> %' + hum + '</div>' +
          '<div class="weather-stat"><i class="fas fa-wind"></i> ' + wind + ' km/s</div>' +
        '</div>';
    })
    .catch(function () {
      box.innerHTML = '<p style="font-size:13px;color:#aaa;padding:8px 0">Hava durumu alınamadı.</p>';
    });
})();
