            (function () {
                function focusable(el) {
                    if (!el) return null;
                    if (!el.hasAttribute('tabindex')) el.setAttribute('tabindex', '-1');
                    return el;
                }
                function findSearchInput() {
                    var panel = document.getElementById('search');
                    if (!panel) return null;
                    return panel.querySelector('input[type="search"], input[type="text"], input:not([type])');
                }
                function handleSkip(e) {
                    var target = e.currentTarget.getAttribute('data-skip-to');

                    if (target === 'search') {
                        e.preventDefault();
                        var trigger = document.querySelector('.search-trigger');
                        if (trigger) trigger.click();
                        setTimeout(function () {
                            var input = findSearchInput();
                            if (input) input.focus();
                            else {
                                var panel = focusable(document.getElementById('search'));
                                if (panel) panel.focus();
                            }
                        }, 150);
                        return;
                    }

                    if (target === 'nav') {
                        e.preventDefault();
                        var navTrigger = document.querySelector('.nav-trigger');
                        if (navTrigger) navTrigger.click();
                        setTimeout(function () {
                            var nav = document.getElementById('primary-nav');
                            if (nav) {
                                var firstLink = nav.querySelector('a, button');
                                if (firstLink) firstLink.focus();
                                else focusable(nav).focus();
                            }
                        }, 150);
                        return;
                    }

                    if (target === 'main') {
                        e.preventDefault();
                        var main = focusable(document.getElementById('main-content'));
                        if (main) main.focus();
                        return;
                    }
                }
                var skipLinks = document.querySelectorAll('.skip-link[data-skip-to]');
                for (var i = 0; i < skipLinks.length; i++) {
                    skipLinks[i].addEventListener('click', handleSkip);
                }
            })();

                    <div class="wrapper">
                        <p><strong>Aile ve Sosyal Hizmetler Bakanımız Mahinur Özdemir Göktaş, ihtiyaç sahibi hanelerdeki öğrencilerin eğitim ihtiyaçlarının karşılanması amacıyla Sosyal Yardımlaşma ve Dayanışma Vakıflarına (SYDV) 2026-2027 eğitim öğretim yılının ilk dönemi için yaklaşık 2 milyar lira kaynak aktardıklarını bildirdi</strong></p>
<p>Bakan Göktaş, yaptığı açıklamada, ihtiyaç sahibi hanelerdeki öğrencilerin eğitim masraflarının karşılanmasına yönelik uyguladıkları eğitim yardımlarına yeni eğitim-öğretim döneminde de devam edeceklerini belirtti.</p>
<p><strong>Öğrencilere kırtasiye, barınma ve ulaşım desteği</strong></p>
<p>Eğitim yardımları kapsamında, öğrencilerin temel okul ihtiyaçlarının karşılanmasına yönelik eğitim materyali yardımlarının yapıldığını aktaran Bakan Göktaş, kırtasiye, okul kıyafeti, ayakkabı ve kışlık kıyafet gibi ihtiyaçların bu kapsamda öğrencilere ulaştırıldığını ifade etti.</p>
<p>Taşımalı eğitim uygulaması kapsamı dışında kalan öğrencilere de barınma, taşıma ve yemek giderlerine yönelik destek sağlandığını kaydeden Bakan Göktaş, lise ve üniversite sınavlarına hazırlanan öğrencilerin sınavlara hazırlık materyallerinin temin edildiğini bildirdi.</p>
<p>Bakan Göktaş, öğrencilerin gezi ve sportif faaliyetlere katılımlarının desteklendiğini, ayrıca yurt içindeki yükseköğretim kurumlarında öğrenim gören ve 3294 sayılı Sosyal Yardımlaşma ve Dayanışmayı Teşvik Kanunu kapsamında bulunan Türk öğrencilere yönelik çeşitli eğitim yardımlarının sürdürüldüğünü belirtti.</p>
<p>Ailelerinden uzakta örgün eğitime devam eden ihtiyaç sahibi öğrencilerin ulaşım giderlerinin de desteklendiğini aktaran Göktaş, öğrencilerin ailelerinin yanına gidip-dönmekte kullanacakları yıllık 2 gidiş-2 dönüş toplam dört bilet hâlindeki Ulaşım Giderlerinin Karşılanmasına Yönelik Destek Programından da yararlanabildiklerini hatırlattı.</p>
<p><strong>“Daha fazla öğrenciye ulaşmayı amaçlıyoruz”</strong></p>
<p>Bakan Göktaş, eğitim yardımlarının temel amacının ihtiyaç sahibi hanelerde bulunan ve öğrenim hayatına devam eden öğrencilerin eğitime yönelik ihtiyaçlarının karşılanması olduğunu vurgulayarak, "Daha fazla öğrenciye ulaşmayı ve yardımların etkinliğini artırmayı amaçlıyoruz." dedi.</p>
<p>Göktaş, 2026 yılının Ocak-Temmuz döneminde Ulaşım Giderlerinin Karşılanmasına Yönelik Destek Programı kapsamında Sosyal Yardımlaşma ve Dayanışmayı Teşvik (SYDT) Fonundan yaklaşık 92 milyon lira kaynak aktarıldığını bildirdi.</p>
<p><strong>Eğitim yardımlarıyla ailelerin yükü hafifletiliyor</strong></p>
<p>İhtiyaç sahibi hanelerin ilköğrenim ve ortaöğrenim düzeyinde örgün eğitime devam eden çocuklarına yönelik eğitim yardımlarında etkinliğin ve verimliliğin artırılmasına önem verdiklerini belirten Bakan Göktaş, bu kapsamda 2021 yılından bu yana SYDT Fonundan SYDV'lere yaklaşık 6 milyar tutarında kaynak aktarıldığını kaydetti.</p>
<p>Bakan Göktaş, aynı amaçla 2026-2027 eğitim öğretim yılının ilk dönemi için SYDV'lere yaklaşık 2 milyar lira tutarında kaynak aktarıldığının altını çizerek, aktarılan kaynakların ihtiyaç sahibi hanelerdeki öğrencilerin eğitim ihtiyaçlarının karşılanması amacıyla ayni ve/veya nakdi yardımların yapılmasında kullanılacağını belirtti.</p>
<p>Göktaş, eğitim yardımlarıyla öğrencilerin eğitim süreçlerine destek olurken, ihtiyaç sahibi ailelerin üzerindeki eğitim giderlerinden kaynaklanan yükün de azaltılmasına katkı sağlandığını ifade etti.</p>

            function debounce(func, wait, immediate) {
                var timeout;
                return function () {
                    var context = this, args = arguments;
                    var later = function () {
                        timeout = null;
                        if (!immediate) func.apply(context, args);
                    };
                    var callNow = immediate && !timeout;
                    clearTimeout(timeout);
                    timeout = setTimeout(later, wait);
                    if (callNow) func.apply(context, args);
                };

                function enhanceSubmenus() {
                    $('li.has-children').each(function () {

                            $trigger.attr('aria-controls', submenuId);

                            var label = $trigger.text().replace(', alt menü içerir', '').trim();

                // Görsel mobil işlev (tıklama) korunur; sadece ekran okuyucu/klavye odağından kaldırılır.
                function hideMenuGoBackFromSR() {
                    var items = document.querySelectorAll('.go-back, .see-all');
                    for (var i = 0; i < items.length; i++) {
                        var li = items[i];
                        if (li.getAttribute('data-sr-hidden') === 'true') continue;
                        li.setAttribute('aria-hidden', 'true');
                        var links = li.querySelectorAll('a');
                        for (var j = 0; j < links.length; j++) {
                            links[j].setAttribute('tabindex', '-1');
                        }
                        li.setAttribute('data-sr-hidden', 'true');
                    }

                        }
                    }
