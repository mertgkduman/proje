    var shareableModelId = 150419;

    function shareCount(id, model, platform, url) {
        fetch("https://www.bursasaati.com.tr/sharecount", {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]')?.getAttribute('content')
            },
            body: JSON.stringify({ id, model, platform, url })
        }).catch(err => console.error('Share count fetch error:', err));

        bumpShareCounter();
    }

    // Paylaşım sayacını yerelde +1 göster. Kalıcı değer edge /v1/hit yanıtından
    // gelir ama o taban 15dk cache'li — kullanıcı kendi paylaşımının sayıya
    // yansıdığını hemen görsün.
    //
    // Sayaç tire (—) ise, yani değer 0 ya da henüz gelmemişse, ilk paylaşım
    // onu "1" yapar. statsRaw dataset'ten okunur (tire metninden sayı
    // üretilemez); yoksa 0 varsayılır.
    function bumpShareCounter() {
        const box = document.querySelector('[data-stats="share"]');
        if (!box) {
            return;
        }
        const target = box.querySelector('[data-stats-value="share"]');
        if (!target) {
            return;
        }
        const current = parseInt(box.dataset.statsRaw || '0', 10) || 0;
        const next = current + 1;
        box.dataset.statsRaw = String(next);
        target.textContent = window.HitManagement
            ? new window.HitManagement().formatShort(next)
            : String(next);
    }

    function goSharePopup(url, title, width = 600, height = 400) {
        const left = (screen.width - width) / 2;
        const top = (screen.height - height) / 2;
        window.open(
            url,
            title,
            `width=${width},height=${height},left=${left},top=${top},resizable=yes,scrollbars=yes`
        );
    }

    async function handleMobileShare(event, title, url) {
        event.preventDefault();

        if (shareableModelId && shareableModelClass) {
            shareCount(shareableModelId, shareableModelClass, 'native_mobile_share', url);
        }

        const isAndroidWebView = navigator.userAgent.includes('Android') && !navigator.share;

        if (isAndroidWebView) {
            window.location.href = 'androidshare://paylas?title=' + encodeURIComponent(title) + '&url=' + encodeURIComponent(url);
            return;
        }

        if (navigator.share) {
            try {
                await navigator.share({ title: title, url: url });
            } catch (error) {
                if (error.name !== 'AbortError') {
                    console.error('Web Share API failed:', error);
                }
            }
        } else {
            alert("Bu cihaz paylaşımı desteklemiyor.");
        }
    }

    function initiateDesktopShare(event, platformOverride = null) {
        event.preventDefault();
        const anchor = event.currentTarget;
        const platform = platformOverride || anchor.dataset.platform;
        const webShareUrl = anchor.href;
        const contentUrl = anchor.dataset.shareUrl || webShareUrl;

        if (shareableModelId && shareableModelClass && platform) {
            shareCount(shareableModelId, shareableModelClass, platform, contentUrl);
        }

        goSharePopup(webShareUrl, platform ? platform.charAt(0).toUpperCase() + platform.slice(1) : "Share");
    }

    function copyURL(event, urlToCopy) {
        event.preventDefault();
        navigator.clipboard.writeText(urlToCopy).then(() => {
            alert('Bağlantı panoya kopyalandı!');
        }).catch(err => {
            console.error('Could not copy text: ', err);
            try {
                const textArea = document.createElement("textarea");
                textArea.value = urlToCopy;
                textArea.style.position = "fixed";
                document.body.appendChild(textArea);
                textArea.focus();
                textArea.select();
                document.execCommand('copy');
                document.body.removeChild(textArea);
                alert('Bağlantı panoya kopyalandı!');
            } catch (fallbackErr) {
                console.error('Fallback copy failed:', fallbackErr);
            }
        });
    }

    function printContent(event) {
        event.preventDefault();

        const triggerElement = event.currentTarget;
        const contextContainer = triggerElement.closest('.infinite-item') || document;

        const header      = contextContainer.querySelector('.post-header');
        const media       = contextContainer.querySelector('.news-section .col-lg-8 .inner, .news-section .col-lg-8 .ratio, .news-section .col-lg-8 iframe');
        const articleBody = contextContainer.querySelector('.article-text');

        if (!header && !media && !articleBody) {
            window.print();
            return;
        }

        let printHtml = '';
        
        if (header) {
            const titleEl = header.querySelector('h1');
            const descEl  = header.querySelector('.description, h2.lead');

            let cleanHeaderHtml = '<div class="printed-header">';
            if (titleEl) cleanHeaderHtml += titleEl.outerHTML;
            if (descEl)  cleanHeaderHtml += descEl.outerHTML;
            cleanHeaderHtml += '</div>';

            printHtml += cleanHeaderHtml;
        }

        if (media) {
            printHtml += media.outerHTML;
        }

        if (articleBody) {
            const articleClone = articleBody.cloneNode(true);
            articleClone.querySelectorAll('.post-flash').forEach(function (el) {
                el.parentNode.removeChild(el);
            });
            printHtml += articleClone.outerHTML;
        }
        const iframe = document.createElement('iframe');
        iframe.style.position = 'fixed';
        iframe.style.right = '0';
        iframe.style.bottom = '0';
        iframe.style.width = '0';
        iframe.style.height = '0';
        iframe.style.border = '0';
        document.body.appendChild(iframe);

        const frameWindow = iframe.contentWindow || iframe;
        const title = document.title || 'Yazdır';

        // Yazdırma belgesi DOM API ile kurulur. Eskiden doc.write ile tam bir HTML
        // belgesi metni yazılıyordu; o metin sayfa kaynağında fazladan head/body
        // etiketleri gibi görünüp SEO denetim araçlarını yanıltıyordu.
        // Yalnız doctype yazılır: standart modda (quirks değil) boş html/head/body oluşur.
        const doc = frameWindow.document;
        doc.open();
        doc.write('<!DOCTYPE html>');
        doc.close();

        doc.documentElement.lang = 'tr';

        const charset = doc.createElement('meta');
        charset.setAttribute('charset', 'utf-8');
        doc.head.appendChild(charset);

        doc.title = title;

        // Sayfanın stilleri kopyalanır; <link>'ler yüklenince yazdırılır (aksi halde çıktı stilsiz kalır)
        const styleLoads = [];
        document.querySelectorAll('link[rel="stylesheet"], style').forEach(function (el) {
            const clone = doc.importNode(el, true);
            if (clone.tagName === 'LINK') {
                styleLoads.push(new Promise(function (resolve) {
                    clone.onload = clone.onerror = resolve;
                }));
            }
            doc.head.appendChild(clone);
        });

        const printStyle = doc.createElement('style');
        printStyle.textContent =
            'html, body { margin: 0; padding: 0; background: #ffffff; }' +
            '.printed-article { margin: 0; padding: 20px; box-shadow: none; background: #ffffff; }';
        doc.head.appendChild(printStyle);

        const wrapper = doc.createElement('div');
        wrapper.className = 'printed-article';
        wrapper.innerHTML = printHtml;
        doc.body.appendChild(wrapper);

        // Stil dosyası takılırsa yazdırma sonsuza dek beklemesin
        const stylesReady = Promise.race([
            Promise.all(styleLoads),
            new Promise(function (resolve) { setTimeout(resolve, 3000); })
        ]);

        stylesReady.then(function () {
            try {
                frameWindow.focus();
                frameWindow.print();
            } finally {
                setTimeout(function () {
                    document.body.removeChild(iframe);
                }, 1000);
            }
        });
    }

<p>Kongrede konuşan Saadet Partisi Yıldırım İlçe Başkanı Nuri Keçeli, kongrelerin teşkilatların azmini tazelediği ve yeni bir dönemin başlatıldığı önemli dönüm noktaları olduğunu söyledi.</p>

<p><img alt="Whatsapp Image 2026 09 27 At 19.00.57 (1)" class="detail-photo img-fluid" height="853" src="https://bursasaaticomtr.teimg.com/bursasaati-com-tr/uploads/2026/09/whatsapp-image-2026-09-27-at-190057-1.jpg" width="1280" /></p>

<h3>“YAŞANABİLİR BİR YILDIRIM HEDEFLİYORUZ”</h3>

<p>Yıldırım’ın farklı bölgelerden gelen insanların birlikte yaşadığı bir kardeşlik merkezi olduğunu belirten Keçeli, ilçenin mevcut sorunlarına dikkat çekti. Keçeli, çarpık kentleşme, altyapı yetersizliği, kentsel dönüşüm mağduriyetleri, gençlerin karşı karşıya kaldığı sosyal sorunlar ve ekonomik sıkıntıların Yıldırım’da ciddi sorun alanları oluşturduğunu ifade etti.</p><div class="advert-paragraph" data-paragraph-advert="121"><div id="ad_121" data-channel="121" data-advert="temedya" data-rotation="120" class="d-none d-sm-flex flex-column align-items-center justify-content-start text-center mx-auto overflow-hidden" data-affix="0" data-bik-reklam="true" data-bik-reklam-id="reklam121n2"></div><div id="ad_121_mobile" data-channel="121" data-advert="temedya" data-rotation="120" class="d-flex d-sm-none flex-column align-items-center justify-content-start text-center mx-auto overflow-hidden" data-affix="0" data-bik-reklam="true" data-bik-reklam-id="reklam121mn2"></div></div>

<p>Saadet Partisi olarak “gelenekten geleceğe yeni bir siyaset anlayışıyla yaşanabilir bir Yıldırım” hedeflediklerini belirten Keçeli, siyaset anlayışlarının sorun üretmek yerine çözüm üretmeye, ahlak ve maneviyat temelinde hizmet etmeye dayandığını söyledi.</p>

<p>Keçeli, “Adil paylaşımı, rüşvetsiz ve şeffaf yönetimi, önce ahlak ve maneviyat ilkesini Yıldırım’ın merkezine koyacağız” dedi.</p>

<p><img alt="Whatsapp Image 2026 09 27 At 18.18.19" class="detail-photo img-fluid" height="720" src="https://bursasaaticomtr.teimg.com/bursasaati-com-tr/uploads/2026/09/whatsapp-image-2026-09-27-at-181819.jpg" width="1280" /></p>

<h3>“YILDIRIM’DAN BAŞLAYARAK DEĞİŞTİRECEĞİZ”</h3>

<p>Saadet Partisi Bursa İl Başkanı Hamza Gürsel ise Türkiye’de ve Bursa’da yaşanan ekonomik ve sosyal sorunlara ilişkin değerlendirmelerde bulundu.</p>

<p>Gürsel, iktidarın politikalarının ekonomik ve sosyal sorunların ortaya çıkmasında etkili olduğunu savunarak, yolsuzluk iddiaları, geçim sıkıntısı, kentleşme ve imar politikaları üzerinden eleştiriler yöneltti.</p>

<p>Bursa ve özellikle Yıldırım’da insanların daha sıkışık yaşam koşullarına mahkûm edildiğini ileri süren Gürsel, şehirlerin insanlara insanca yaşayabilecekleri koşulları sunması gerektiğini söyledi.</p>

<p><img alt="Whatsapp Image 2026 09 27 At 18.18.18" class="detail-photo img-fluid" height="960" src="https://bursasaaticomtr.teimg.com/bursasaati-com-tr/uploads/2026/09/whatsapp-image-2026-09-27-at-181818.jpg" width="1280" /></p>

<h3>“İNSANLARIN DERTLERİYLE DERTLENECEĞİZ”</h3>

<p>Siyasetin insanlarla iletişim kurarak ve onların sorunlarını dinleyerek yapılması gerektiğini belirten Gürsel, Saadet Partisi teşkilatlarının vatandaşlara ulaşırken kavga ve çatışma yerine diyalog anlayışını benimseyeceğini söyledi.</p>

<p>Gürsel, Yıldırım teşkilatının kısıtlı imkânlarla yürüttüğü çalışmalara da dikkat çekerek, mahalle ve sivil toplum çalışmaları başta olmak üzere teşkilat mensuplarının özverili çalışmalarından dolayı teşekkür etti.</p>

<p><img alt="Whatsapp Image 2026 09 27 At 19.00.56" class="detail-photo img-fluid" height="1919" src="https://bursasaaticomtr.teimg.com/bursasaati-com-tr/uploads/2026/09/whatsapp-image-2026-09-27-at-190056.jpg" width="1280" /></p>

<h3>“TOPLUMDAKİ UMUDU BÜYÜTÜYORUZ”</h3>

<p>Saadet Partisi Genel Başkanı Mahmut Arıkan da kongrede yaptığı konuşmada, toplumdaki en önemli sorunlardan birinin geleceğe ilişkin umutsuzluk olduğunu söyledi.</p>

<p>Üye sayısındaki artışın tek başına yeterli olmadığını belirten Arıkan, siyasette topluma yeniden umut kazandırmanın önemine dikkat çekti. Saadet Partisi teşkilatlarının sınırlı imkânlarla çalışmalarını sürdürdüğünü ifade eden Arıkan, ortaya konulan performansın toplumdaki umudu büyüttüğünü söyledi.</p>

<p><img alt="Whatsapp Image 2026 09 27 At 19.00.57" class=" detail-photo img-fluid" height="853" src="https://bursasaaticomtr.teimg.com/bursasaati-com-tr/uploads/2026/09/whatsapp-image-2026-09-27-at-190057.jpg" width="1280" /></p>

<p>Arıkan, “Hazine imkânlarıyla değil, inanmış teşkilatlarımızla, Allah ne der diye çalışan teşkilatlarımızla, imanla ve azimle bu çalışmaları ortaya koyuyoruz” ifadelerini kullandı.</p>
