function reaksiyonlariYukle() {
    var el = document.getElementById('reaksiyon-alani');
    if (!el) return;
    var id = el.getAttribute('data-id');
    var tur = el.getAttribute('data-tur');
    var grid = document.getElementById('reaksiyon-grid');
    var xhr = new XMLHttpRequest();
    xhr.open('GET', REAKSIYON_AJAX + '?action=reaksiyon_getir&icerik_id=' + id + '&icerik_tur=' + tur);
    xhr.onload = function() {
        if (xhr.status !== 200) return;
        var res = JSON.parse(xhr.responseText);
        if (!res.success) return;
        renderReaksiyonlar(grid, res.sayilar, res.kullanici_secim, id, tur);
    };
    xhr.send();
}
function renderReaksiyonlar(grid, sayilar, secim, id, tur) {
    var html = '';
    for (var i = 0; i < emojiListesi.length; i++) {
        var e = emojiListesi[i];
        var aktif = secim === e.tip ? ' aktif' : '';
        var sayi = sayilar[e.tip] || 0;
        html += '<div class="reaksiyon-item' + aktif + '" onclick="reaksiyonVer(' + e.tip + ')" title="' + e.ad + '">';

        html += '<div class="reaksiyon-cizgi"></div>';
        html += '<span class="reaksiyon-emoji">' + e.emoji + '</span>';
        html += '</div>';
    }
    grid.innerHTML = html;
}

    var el = document.getElementById('reaksiyon-alani');
    if (!el) return;
    var id = el.getAttribute('data-id');
    var tur = el.getAttribute('data-tur');
    var grid = document.getElementById('reaksiyon-grid');

    xhr.open('POST', REAKSIYON_AJAX + '?action=reaksiyon_ver');
    xhr.setRequestHeader('Content-Type', 'application/x-www-form-urlencoded');
    xhr.onload = function() {
        if (xhr.status !== 200) return;
        var res = JSON.parse(xhr.responseText);
        if (!res.success) return;

    };
    xhr.send('icerik_id=' + id + '&icerik_tur=' + tur + '&emoji_tip=' + emojiTip);
}

    function avatarRenk(isim) {
        var hash = 0;
        for (var i = 0; i < isim.length; i++) hash = isim.charCodeAt(i) + ((hash << 5) - hash);
        return avatarRenkler[Math.abs(hash) % avatarRenkler.length];
