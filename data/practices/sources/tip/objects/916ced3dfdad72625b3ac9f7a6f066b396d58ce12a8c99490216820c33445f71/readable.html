(function(){

      results.innerHTML = '';
      return;

  async function search(q){
    try{
      const url = endpointBase + (endpointBase.includes('?') ? '&' : '?') + '__ajax=search&q=' + encodeURIComponent(q);
      const res = await fetch(url, {credentials:'same-origin'});
      if (!res.ok) throw new Error('HTTP '+res.status);
      const data = await res.json();
      const items = Array.isArray(data.items) ? data.items : [];
      if (!items.length) {
        results.innerHTML = '<div class="sr-empty">☹️ Maalesef aradığın kelimeye ait bir sonuç bulamadık.</div>';
        results.classList.add('show');
        return;
      }
      results.innerHTML = items.map(it=>{
        const thumb = it.thumb ? it.thumb : 'https://tip.org.tr/wp-content/uploads/2023/05/tip-logo-1600x900-2-scaled.jpg';
        const date  = it.date ? new Date(it.date.replace(' ','T')).toLocaleDateString('tr-TR') : '';
        const badge = it.label ? `<span class="sr-badge">${escapeHtml(it.label)}</span>` : '';
        const snippet = (it.snippet||'').trim();

              <h4 class="sr-title">

              ${date ? `<small style="color:#666">${date}</small>` : ''}
            </span>
          </a>`;
      }).join('');

    } catch(e){

                <!-- wp:paragraph -->
<p></p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Ülkemizin corona virüsü tehdidi altında olduğu resmi olarak bir vakanın saptanmasıyla birlikte netleşmiştir. Sağlık Bakanlığı’nca yapılan açıklamadan bu yana tüm yurttaşlarımıza alabilecekleri kişisel önlemler bildirilmektedir. Bu önlemler arasında en yaşamsal olanları;<br />-Hapşırma, aksırma, öksürme sırasında kağıt mendil kullanmak veya kolla ağzı ve burnu kapamak,<br />-Elleri su ve sabunla düzenli olarak yıkamak, kirli ellerle ağız, burun ve göze dokunmamak,<br />-Geleneksel selamlaşma biçimlerimiz olan tokalaşma, öpüşme ve sarılmadan bir süre kaçınmaktır.<br />Tüm yurttaşlarımızı bu önlemleri almaya çağırıyoruz.</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Ancak dünya geneline yayılan bu ciddi hastalığa karşı kişisel çabalar yeterli değildir. Devletin tüm yetkili kurumları hiç vakit kaybetmeden aşağıdaki önlemleri almalıdır.<br />1- Corona virüsüne karşı en yüksek risk grubunda olan, ölümlerin en sık görüldüğü 60 yaş üstü ve kronik hastalığı olan yurttaşlara kullandıkları ilaçlar ücretsiz verilmelidir.<br />2- Hastalığın artan bir hızla yayıldığı ve 14 günlük bir kuluçka süresi olduğu bilinmektedir. Özellikle kalabalık ortamlarda bulaşmayı önlemek için tüm okullar ve üniversiteler eğitime en az iki hafta ara vermelidir.<br />3- Aynı nedenle kalabalık ortamlarda çalışan emekçilerin tümüne en az iki hafta ücretli idari izin verilmelidir.<br />4- Corona virüsünden etkilenmeye en açık olanlar hastalarla sürekli temas halindeki sağlık emekçileridir. Sağlık emekçilerinin hastalıktan korunması, yurttaşlara en iyi biçimde sağlık hizmeti sunabilmesi için gerekli tüm ilaç, malzeme ve donanımlar sağlanmalı, sağlık emekçilerinin aşırı yoğunluktan olumsuz etkilenmemesi için iş rotasyonları yeniden düzenlenmelidir.<br />5- Kamu hizmeti veren tüm kurumlar dezenfeksiyon ve korunma amaçlı malzemelerle donatılmalıdır.<br />6- Dezenfeksiyon ve korunma amaçlı temel malzemeler karaborsa ve vurgunculuğa karşı halka ücretsiz sağlanmalıdır.<br />7- Borçlarından ötürü elektrik ve suyu kesilmiş yurttaşların kişisel hijyenlerini sağlayabilmeleri için bu hizmetlere erişimi sağlanmalıdır.<br />8- Salgının yaratacağı ekonomik yük, kamu ya da özel sektörde çalışan işçiler, emekçiler, emekliler, kent merkezlerinde ya da kırsal kesimde kendi işinin sahibi küçük işletme sahiplerince değil devlet tarafından karşılanmalıdır. Bu yük ek vergilerle halkın sırtına kesinlikle bindirilmemelidir.<br />9- Bugün ve gelecekte yaşanabilecek her türlü hastalık, salgın, vb. durumlara karşı halk sağlığını korumanın tek yolu olarak piyasacı anlayış terk edilmelidir.</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Karşı karşıya kaldığımız halk sağlığına yönelik bu büyük tehdit ancak ve ancak yetkililerin bilimsel uyarıları dikkate alarak atacakları adımlarla ve hastalığın seyriyle ilgili gerçekleri halktan saklamamakla ortadan kaldırılabilir.</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>Tüm yurttaşlarımızı kamu yararına toplumcu sağlık mücadelesini yükseltmeye çağırıyoruz.<br>TÜRKİYE İŞÇİ PARTİSİ</p>

    }
    function hideLoading(){
      if (!load) return;
      load.style.opacity='0';
      load.style.pointerEvents='none';
      setTimeout(()=>{ if(load) load.style.display='none'; }, 250);
    }

    function openOverlay(){
      ov.classList.remove('hidden');
      ov.setAttribute('aria-hidden','false');
      document.documentElement.classList.add('reader-open'); // scroll kilidi için aynı sınıf

      // src’yi ilk açılışta set et (performans)
      if (!srcSet){
        showLoading();
        setOverlayPage(1);
        srcSet = true;
      } else {
        // tekrar açılışta kısa loading göstermeyelim
        hideLoading();
      }
    }

    async function closeOverlay(){
      ov.classList.add('hidden');
      ov.setAttribute('aria-hidden','true');
      document.documentElement.classList.remove('reader-open');
      if (document.fullscreenElement) { try { await document.exitFullscreen(); } catch(_){ } }
    }

    btn.addEventListener('click', (e)=>{
      e.preventDefault();
      e.stopPropagation();
      openOverlay();
    });

    close?.addEventListener('click', closeOverlay);

    // overlay dışına tıklayınca kapat (toolbar/body içi hariç)
    ov.addEventListener('click', (e)=>{
      const t = e.target;
      if (t === ov) closeOverlay();
    });

      if (e.key === 'Escape' && ov.getAttribute('aria-hidden') === 'false') closeOverlay();
    });

    fsBtn?.addEventListener('click', async ()=>{
      try{
        if (document.fullscreenElement) await document.exitFullscreen();
        else await ov.requestFullscreen?.();
      }catch(_){}
    });
    ovPrev?.addEventListener('click', ()=>{ showLoading(); setOverlayPage(ovPage - 1); });
    ovNext?.addEventListener('click', ()=>{ showLoading(); setOverlayPage(ovPage + 1); });
    if (ovBody) {
      let sX = 0, sY = 0;
      ovBody.addEventListener('touchstart', (e)=>{
        if (!e.touches || !e.touches[0]) return;
        sX = e.touches[0].clientX;
        sY = e.touches[0].clientY;
      }, {passive:true});
      ovBody.addEventListener('touchend', (e)=>{
        if (!e.changedTouches || !e.changedTouches[0]) return;
        const eX = e.changedTouches[0].clientX;
        const eY = e.changedTouches[0].clientY;
        const dx = eX - sX;
        const dy = eY - sY;
        if (Math.abs(dx) > 55 && Math.abs(dx) > Math.abs(dy)) {
          showLoading();
          setOverlayPage(ovPage + (dx < 0 ? 1 : -1));
        }
      }, {passive:true});
    }

    frame.addEventListener('load', hideLoading, { once:false });
  })();

  (function(){
    const btn   = document.getElementById("shareBtn_81");
    const menu  = document.getElementById('shareMenu');
    if (!btn || !menu) return;

    const pageTitle = "Salgına Karşı Kamusal Çözüm!";
    const pageUrl   = "https:\/\/tip.org.tr\/parti-aciklamalari\/salgina-karsi-kamusal-cozum";

    function tryWebShare(){
      if (navigator.share){
        navigator.share({ title: pageTitle || document.title, url: pageUrl })
          .catch(()=>{ toggleMenu(true); }); // kullanıcı paylaşımı iptal ederse fallback göster
        return true;
      }
      return false;
    }

    function placeMenu(){
      const prevDisplay = menu.style.display;
      const prevVis     = menu.style.visibility;
      menu.style.visibility = 'hidden';
      menu.style.display    = 'block';

      const r   = btn.getBoundingClientRect(); // viewport koordinatları
      const mh  = menu.offsetHeight;
      const mw  = menu.offsetWidth;

      const spaceBelow = window.innerHeight - r.bottom;
      const openUp     = spaceBelow < (mh + 12);

      let top  = openUp ? (r.top - mh - 8) : (r.bottom + 8);
      let left = r.right - mw; // sağa hizala

      const PAD = 12;
      left = Math.min(window.innerWidth - PAD - mw, Math.max(PAD, left));
      if (openUp && top < PAD) top = PAD;
      if (!openUp && (top + mh > window.innerHeight - PAD)) top = window.innerHeight - PAD - mh;

      menu.style.top  = top  + 'px';
      menu.style.left = left + 'px';

      menu.style.display    = prevDisplay || 'none';
      menu.style.visibility = prevVis     || '';
    }

    function toggleMenu(show){
      const willShow = (typeof show==='boolean') ? show : menu.getAttribute('aria-hidden') !== 'false';
      if (willShow) placeMenu();
      menu.setAttribute('aria-hidden', willShow ? 'false' : 'true');
      btn.setAttribute('aria-expanded', willShow ? 'true' : 'false');
    }

    btn.addEventListener('click', function(e){
      e.preventDefault();
      e.stopPropagation();
      if (!tryWebShare()){
        toggleMenu();
      }
    });

    document.addEventListener('click', function(e){
      if (!menu || menu.getAttribute('aria-hidden') === 'true') return;
      const t = e.target;
      if (t === btn || btn.contains(t) || menu.contains(t)) return;
      toggleMenu(false);
    });

    document.addEventListener('keydown', function(e){
      if (e.key === 'Escape') toggleMenu(false);
    });

    ['scroll','resize','orientationchange'].forEach(ev=>{
      window.addEventListener(ev, ()=>{
        if (menu.getAttribute('aria-hidden') === 'false') placeMenu();
      }, {passive:true});
    });

    const copyBtn = menu.querySelector('.copy-link-btn');
    copyBtn?.addEventListener('click', async ()=>{
      try{
        await navigator.clipboard.writeText(pageUrl);
        copyBtn.innerHTML = '<i class="fa-solid fa-check"></i> Kopyalandı';
        setTimeout(()=>{ copyBtn.innerHTML = '<i class="fa-regular fa-copy"></i> Bağlantıyı kopyala'; }, 1600);
      }catch(_){
        const ta = document.createElement('textarea');
        ta.value = pageUrl; document.body.appendChild(ta); ta.select();
        try{ document.execCommand('copy'); }catch(_){}
        document.body.removeChild(ta);
      }
    });
  })();
